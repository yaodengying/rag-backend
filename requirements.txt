# -*- coding: utf-8 -*-
# =============================================================================
# rag-backend 依赖清单
#
# 技术栈：FastAPI + LangChain 1.2 + LangGraph 1.2 + Milvus + RAG
#
# 说明：
#   1. LangChain 1.x 已拆分。核心只有三个包：langchain / langchain-core / langgraph。
#      其余能力（Milvus、OpenAI 兼容接口等）走独立的 provider 包。
#   2. 本项目锁定 LangChain 1.2 系（>=1.2,<1.3），避免 1.3/1.4 的行为漂移。
#      注意：langchain 与 langgraph 的主版本号并不同步。
#      langgraph 1.2.x 依赖 langchain-core>=1.4.7，
#      因此 langchain-core 不能跟着钉在 1.2 系（那样必然解析失败）。
#      已实测：本清单在 Python 3.14 下可完整解析。
#   3. 向量库用 Milvus，直接对接 pymilvus（MilvusClient）：
#      混合检索要同时用到原生全文检索与 BM25 Function，langchain-milvus
#      的封装拿不到逐路分数、也建不出这类 schema，故不经 LangChain 集成层。
#   4. LLM 走 OpenAI 兼容协议（本机 Ollama /v1），不绑定云端厂商 SDK。
#   5. Embedding / Rerank 走本地 BGE 权重目录（FlagEmbedding 官方推理库），不联网。
# =============================================================================

# ---------- Web 框架 ----------
fastapi>=0.115,<1.0
uvicorn[standard]>=0.32
python-multipart>=0.0.12

# ---------- 配置与校验 ----------
pydantic>=2.9
pydantic-settings>=2.6

# ---------- LangChain 1.2 核心 ----------
# 注意：langchain-core 是传递依赖，由 langchain / langgraph 共同决定版本，
# 不要在此单独钉版本，否则会与 langgraph 冲突
# （实测：langgraph 1.2.x 要求 langchain-core>=1.4.7，与「core 1.2 系」不兼容）。
langchain>=1.2,<1.3
langgraph>=1.2,<1.3

# ---------- LangChain 集成 ----------
# 文档加载器（PyPDFLoader / Docx2txtLoader / TextLoader）目前仍在这个包里。
# 注意：langchain-community 已进入 sunset（导入时会打 DeprecationWarning），
# 官方建议迁往独立集成包；但截至本次实测，PDF/Text 还没有可用的独立包
# （langchain-pypdf 不存在，langchain-unstructured 1.0.1 只包 unstructured 那一路）。
# 迁移动向见 https://github.com/langchain-ai/langchain-community/issues/674
langchain-community>=0.4
langchain-openai>=1.0          # OpenAI 兼容协议（指向本机 Ollama /v1）
langchain-text-splitters>=1.0  # 递归字符切分

# ---------- 向量库 ----------
pymilvus>=2.6

# ---------- 向量化 / 重排（本地 BGE 权重，CPU 推理）----------
# BGE 是 BAAI 自家模型，FlagEmbedding 是其官方推理库：直接基于 transformers 的
# AutoModel/AutoTokenizer，不经过 sentence-transformers 这一层。
# 应用代码只 import FlagEmbedding（见 app/infrastructure/embedding.py 与 rerank.py）。
FlagEmbedding>=1.3
# 不套 LangChain 的 Embeddings 抽象：langchain-community / langchain-huggingface 的
# HF 封装在 __init__ 里 import sentence_transformers（本项目已卸载）且已 deprecated，
# 套一层只有命名收益。向量化直接对接 FlagEmbedding，端口 EmbedderPort 保持框架无关。

# 注意：以下两个包**故意不在此钉版本**——它们是 FlagEmbedding 声明的运行时依赖，
# 由 pip 按 FlagEmbedding 的要求解析，直接钉反而会互相打架：
#   FlagEmbedding 1.4.2 要求 torch>=1.6.0、transformers<6.0.0,>=4.44.2
#     - torch 是**硬依赖，无法卸载**（transformers / accelerate / peft 也都需要它）；
#     - sentence-transformers 虽被 FlagEmbedding 写在依赖里，但代码中零 import
#       （全库仅注释与链接提及），故本项目不直接引它，环境里也可安全移除。
#   transformers>=5 需要 torch>=2.6（用到 torch.accelerator），
#   所以实际解出的 torch 不会低于 2.6，勿手工回退到 2.5.x。

# ---------- 文档加载（各 Loader 的底层库，app/infrastructure/loading.py）----------
pypdf>=5.0         # PyPDFLoader 的解析后端，逐页提取、页码准确
docx2txt>=0.9      # Docx2txtLoader 的解析后端，段落 + 表格单元格
python-pptx>=1.0   # 自研 PptxLoader（langchain-core 的 BaseLoader）的解析后端
# 说明：不再直接用 python-docx 做解析。它只能按段落遍历，表格要手写循环，
# 而 Docx2txtLoader 两者的抽取都已覆盖（实测表格单元格不丢）。
# python-docx 仍保留在「开发期」一节，仅用于生成测试样例。

# ---------- 元数据持久化 ----------
sqlalchemy>=2.0.36

# ---------- 工具 ----------
httpx>=0.27        # scripts/ask.py 与 verify_upload_flow.py 的 HTTP 客户端
numpy>=1.26        # scripts/check_models.py 用它对数值做断言
